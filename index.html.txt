    <script>
        const RATE = 1.95583;
        let resultCurrency = 'BGN';

        // Генериране на фон
        const bgGrid = document.getElementById('bg-grid');
        for (let i = 0; i < 24; i++) {
            const div = document.createElement('div');
            div.className = 'symbol';
            div.innerText = i % 2 === 0 ? 'лв' : '€';
            div.style.animationDelay = `${Math.random() * 5}s`;
            bgGrid.appendChild(div);
        }

        // Логика за изчисление
        const inputs = ['billAmount', 'billCurrency', 'paidBGN', 'paidEUR'];
        inputs.forEach(id => document.getElementById(id).addEventListener('input', calculate));

        function setResultCurrency(curr) {
            resultCurrency = curr;
            document.getElementById('btnBGN').className = curr === 'BGN' ? 'px-4 py-1 rounded-lg text-sm font-bold btn-beige shadow-sm' : 'px-4 py-1 rounded-lg text-sm font-bold';
            document.getElementById('btnEUR').className = curr === 'EUR' ? 'px-4 py-1 rounded-lg text-sm font-bold btn-beige shadow-sm' : 'px-4 py-1 rounded-lg text-sm font-bold';
            calculate();
        }

        function calculate() {
            const bill = parseFloat(document.getElementById('billAmount').value) || 0;
            const billCurr = document.getElementById('billCurrency').value;
            const paidBGN = parseFloat(document.getElementById('paidBGN').value) || 0;
            const paidEUR = parseFloat(document.getElementById('paidEUR').value) || 0;

            // Всичко към BGN за база
            const billInBGN = billCurr === 'EUR' ? bill * RATE : bill;
            const totalPaidInBGN = paidBGN + (paidEUR * RATE);
            const changeInBGN = totalPaidInBGN - billInBGN;

            const errorBox = document.getElementById('errorBox');
            const resultSection = document.getElementById('resultSection');

            if (totalPaidInBGN < billInBGN && billInBGN > 0) {
                errorBox.classList.remove('hidden');
                resultSection.classList.add('hidden');
            } else if (billInBGN > 0) {
                errorBox.classList.add('hidden');
                resultSection.classList.remove('hidden');
                displayResults(changeInBGN);
            } else {
                resultSection.classList.add('hidden');
            }
        }

        function displayResults(bgnChange) {
            const eurChange = bgnChange / RATE;
            
            if (resultCurrency === 'BGN') {
                document.getElementById('mainChange').innerText = `${bgnChange.toFixed(2)} лв`;
                document.getElementById('secondaryChange').innerText = `еквивалент: ${eurChange.toFixed(2)} €`;
            } else {
                document.getElementById('mainChange').innerText = `${eurChange.toFixed(2)} €`;
                document.getElementById('secondaryChange').innerText = `еквивалент: ${bgnChange.toFixed(2)} лв`;
            }
        }

        // PWA ЛОГИКА - Динамично генериране на Manifest и Service Worker
        (function setupPWA() {
            // 1. Създаване на икона чрез Canvas
            const canvas = document.createElement('canvas');
            canvas.width = 512; canvas.height = 512;
            const ctx = canvas.getContext('2d');
            ctx.fillStyle = '#4b0082';
            ctx.beginPath();
            ctx.roundRect(0, 0, 512, 512, 100);
            ctx.fill();
            ctx.fillStyle = '#f5f5dc';
            ctx.font = 'bold 120px Comfortaa, sans-serif';
            ctx.textAlign = 'center';
            ctx.fillText('€ ⇄ лв', 256, 280);
            const iconUrl = canvas.toDataURL('image/png');

            // 2. Web Manifest
            const manifest = {
                name: 'DualPay Change',
                short_name: 'DualPay',
                start_url: '.',
                display: 'standalone',
                background_color: '#e0eafc',
                theme_color: '#4b0082',
                icons: [{ src: iconUrl, sizes: '512x512', type: 'image/png' }]
            };
            const manifestBlob = new Blob([JSON.stringify(manifest)], {type: 'application/json'});
            const manifestURL = URL.createObjectURL(manifestBlob);
            const link = document.createElement('link');
            link.rel = 'manifest';
            link.href = manifestURL;
            document.head.appendChild(link);

            // 3. Service Worker
            const swCode = `
                self.addEventListener('install', e => self.skipWaiting());
                self.addEventListener('fetch', e => e.respondWith(fetch(e.request).catch(() => caches.match(e.request))));
            `;
            const swBlob = new Blob([swCode], {type: 'application/javascript'});
            const swURL = URL.createObjectURL(swBlob);
            if ('serviceWorker' in navigator) {
                navigator.serviceWorker.register(swURL);
            }
        })();
    </script>
